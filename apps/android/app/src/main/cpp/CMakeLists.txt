cmake_minimum_required(VERSION 3.22.1)
project(jarvis_llm LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(LLAMA_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/llama.cpp")
if(NOT EXISTS "${LLAMA_SOURCE_DIR}/include/llama.h")
  message(FATAL_ERROR "Pinned llama.cpp missing. Run tools/prepare_llama_cpp.py")
endif()

set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
# ARM64 Android: pinned llama.cpp includes KleidiAI AArch64 kernels.
# Keep x86_64 emulator/other targets on the portable baseline. No host-native architecture flags.
if(ANDROID_ABI STREQUAL "arm64-v8a")
  set(GGML_CPU_KLEIDIAI ON CACHE BOOL "" FORCE)
else()
  set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
endif()
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)

add_subdirectory("${LLAMA_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/upstream")
add_library(jarvis_llm SHARED jarvis_llm.cpp)
target_include_directories(jarvis_llm PRIVATE "${LLAMA_SOURCE_DIR}/include")
target_compile_options(jarvis_llm PRIVATE -fvisibility=hidden -fno-exceptions)
target_link_libraries(jarvis_llm PRIVATE llama android log)
